\section{Logistic Regression}

\begin{frame}{Logistic Regression for Classification}
    \begin{greenblock}{Definition}
        \begin{itemize}
            \item Supervised learning
            \item Predict categorical outcomes
                \begin{itemize}
                    \item Binary classification: yes/no, 1/0
                    \item Multiple classes
                \end{itemize}
        \end{itemize}
    \end{greenblock}
    \begin{columns}
        \column{\kitthreecolumns}
        \begin{greenblock}{Goal}
            Predict the probability that a sample $x$ belongs to class 1
            \begin{equation*}
                P(y = 1 \mid x)
            \end{equation*}
            \begin{itemize}
                \item $x \in \mathbb{R}$: a single input feature
                \item $y \in \{0, 1\}$: target class label
            \end{itemize}
        \end{greenblock}

        \column{\kitthreecolumns}
        \begin{greenblock}{Decision Rule}
            Binary classification with decision boundary at $0.5$
            \begin{equation*}
                \hat{y} =
                \begin{cases}
                1 & \text{if } P(y = 1 \mid x) \geq 0.5 \\
                0 & \text{otherwise}
                \end{cases}
            \end{equation*}
        \end{greenblock}
    \end{columns}
\end{frame}

\begin{frame}{Sigmoid Function}
    \begin{columns}
        \column{\kitthreecolumns}
        \begin{greenblock}{a.k.a logistic function}
            \begin{equation*}
                P(y = 1 \mid x) = \frac{1}{1 + \exp[-(wx + b)]}
            \end{equation*}
            \begin{itemize}
                \item $w$: weight (slope)
                \item $b$: bias (intercept)
            \end{itemize}
        \end{greenblock}

        \column{\kitthreecolumns}
        \includegraphics[width=\textwidth]{images/Exam_pass_logistic_curve.png}
    \end{columns}
\end{frame}

\begin{frame}{Multi-class Classification: Multinomial Regression}
    \begin{columns}
        \column{\kitthreecolumns}
        \begin{greenblock}{Example: Iris Dataset}
            \begin{itemize}
                \item $n = 150$ samples
                \item $d = 4$ features (sepal length/width, petal length/width)
                \item $K = 3$ classes (setosa, versicolor, virginica)
                \item $\boldsymbol{x}^{(i)} \in \mathbb{R}^2$ is the feature
                    vector for the $i^{th}$ sample
                \item $\boldsymbol{x}^{(i)} \in \{0, 1, 2\}$ is the class label
                    for the $i^{th}$ sample
            \end{itemize}
        \end{greenblock}

        \column{\kitthreecolumns}

        \begin{greenblock}{Multinomial Regression (softmax)}
            \textbf{Idea:} learn a single model with one set of weights per
            class, all weights are learned simultaneously
            \begin{equation*}
                P(y = k \mid \boldsymbol{x}) = \frac{\exp[-(\boldsymbol{w}_k^T
                \boldsymbol{x})]}{\sum_j \exp[-(\boldsymbol{w}_j^T
            \boldsymbol{x})]}
            \end{equation*}
            \begin{itemize}
                \item Weights: $W \in \mathbb{R}^{3 \times 4}$, where row $k$ is
                    $\boldsymbol{w}_k^T$
                \item Bias: $\boldsymbol{b} \in \mathbb{R}^3$
                \item $i \in \{1, ..., 150\}$
                \item Result: single vector of three class probabilities,
                    summing to 1
            \end{itemize}
        \end{greenblock}
    \end{columns}
\end{frame}

\begin{frame}{Multi-class Classification: One-vs-Rest}
    \begin{columns}
        \column{\kitthreecolumns}
        \begin{greenblock}{Example: Iris Dataset}
            \begin{itemize}
                \item $n = 150$ samples
                \item $d = 4$ features (sepal length/width, petal length/width)
                \item $K = 3$ classes (setosa, versicolor, virginica)
                \item $\boldsymbol{x}^{(i)} \in \mathbb{R}^2$ is the feature
                    vector for the $i^{th}$ sample
                \item $\boldsymbol{x}^{(i)} \in \{0, 1, 2\}$ is the class label
                    for the $i^{th}$ sample
            \end{itemize}
        \end{greenblock}

        \column{\kitthreecolumns}

        \begin{greenblock}{One-vs-Rest (sigmoid)}
            \textbf{Idea:} train $K$ separate binary classifiers, one for each
            class $k$, to distinguish class $k$ from all others
            \begin{equation*}
                P(y = k \mid \boldsymbol{x}^{(i)}) = \frac{1}{1 +
                \exp[-(\boldsymbol{w}_k^T \boldsymbol{x}^{(i)} + b_k)]}
            \end{equation*}

            \begin{itemize}
                \item Weights: $\boldsymbol{w}_k \in \mathbb{R}^4$
                \item Bias: $b_k \in \mathbb{R}$
                \item $i \in \{1, ..., 150\}$
                \item Result: three scalar probabilities per sample
            \end{itemize}
        \end{greenblock}
    \end{columns}
\end{frame}

\begin{frame}{Exercise: One-vs-Rest Regression}
    \begin{columns}
        \column{\kitthreecolumns}
        \begin{greenblock}{Training}
            \textbf{Create new labels}
            \begin{equation*}
                y^{(i)} =
                \begin{cases}
                1 & \text{if original class is}~k\\
                0 & \text{otherwise}
                \end{cases}
            \end{equation*}
            \textbf{Cross-entropy loss function}
            \begin{equation*}
                \mathcal{L}_k = - \sum_{i=1}^{n} \left[ y_k^{(i)}
                \log(\hat{y}_k^{(i)}) + (1-y_k^{(i)}) \log(1-\hat{y}_k^{(i)}) \right]
            \end{equation*}
            \begin{itemize}
                \item Typical for classification
                \item Can be optimized well due to global minimum
            \end{itemize}
        \end{greenblock}

        \column{\kitthreecolumns}
        \begin{greenblock}{Prediction}
            \textbf{Predictor for each class $k$}
            \begin{equation*}
                \hat{y}_k = \frac{1}{1 + \exp[-(\mathbf{w}_k^T \mathbf{x} +
                b_k)]}
            \end{equation*}
            \textbf{Select classifier with highest probability}
            \begin{equation*}
                \hat{y} = \argmax_k \hat{y}_k
            \end{equation*}
            
        \end{greenblock}
    \end{columns}
\end{frame}


\section{Support Vector Machines}

\begin{frame}{Support Vector Classifiers -- Concepts}
    \begin{columns}
        \column{\kitthreecolumns}
        \begin{greenblock}{Support Vector Classifier (SVC)}
            \begin{itemize}
                \item Linear classifier
                \item Best possible hyperplane that divides data
                \item Idea: find the maximum margin
            \end{itemize}
        \end{greenblock}

        \begin{greenblock}{Margin}
            \begin{itemize}
                \item Distance between the hyperplane and the closest data points
                \item Closest points: support vectors
                \item Hard-margin: no misclassifications allowed
                \item Soft-margin: slack variables allowing misclassifications
                    (regularization parameter)
            \end{itemize}
        \end{greenblock}

        \column{\kitthreecolumns}
        \begin{center}
            \includegraphics[width=0.8\textwidth]{images/SVM.png}
        \end{center}
    \end{columns}
\end{frame}

\begin{frame}{Support Vector Machines (SVMs)}
    \begin{columns}
        \column{\kitthreecolumns}
        \begin{greenblock}{Nonlinear Classifier}
            \begin{itemize}
                \item Start with the low-dimensional data
                \item Transform the data to a higher dimension
                \item Use a linear SVC to separate the data
            \end{itemize}
        \end{greenblock}

        \begin{greenblock}{Kernel Trick}
            \begin{itemize}
                \item Kernel trick: calculate the relationship between different
                    points in a higher dimension \textbf{without} the actual
                    transformation of the data points
                \item Kernel function: computes the inner dot product between
                    two points on the higher-dimensional space
            \end{itemize}
        \end{greenblock}

        \column{\kitthreecolumns}
        \begin{center}
            \includegraphics[width=\textwidth]{images/Non-linear-classifier-using-Kernel-trick-16.png}
        \end{center}
    \end{columns}
\end{frame}

\begin{frame}{Polynomial Kernel}
    \begin{columns}
        \column{\kitthreecolumns}
        \begin{greenblock}{Equation}
            \begin{equation*}
                K(x, x^\prime) = (1 + \langle x, x^\prime \rangle )^d
            \end{equation*}
            \begin{itemize}
                \setlength\itemsep{2em}
                \item Degree $d$ (hyperparameter)
                \item Dot product between the two features $x$ and $x^\prime$: $\langle
                    x, x^\prime \rangle$
            \end{itemize}
        \end{greenblock}

        \column{\kitthreecolumns}
        \begin{center}
            \includegraphics[width=0.7\textwidth]{images/Polynomial_kernel.png}
        \end{center}
    \end{columns}
\end{frame}

\begin{frame}{Radial Basis Function (RBF)}
    \begin{columns}
        \column{\kitthreecolumns}
        \begin{greenblock}{Equation}
            \begin{equation*}
                K(x, x^\prime) = \exp(-\gamma \Vert x-x^\prime \Vert^2)
            \end{equation*}

            \begin{itemize}
                \setlength\itemsep{2em}
                \item a.\,k.\,a. radial kernel or Gaussian kernel
                \item Squared Euclidean distance: $\Vert x-x^\prime \Vert = \sum_i (x_i
                    - x_i^\prime)^2$
                \item Width of Gaussian: $\gamma > 0$ (how quickly does the similarity
                    decay)
            \end{itemize}
        \end{greenblock}

        \column{\kitthreecolumns}
        \begin{greenblock}{Interpretation}
            \begin{itemize}
                \item Technically has infinite dimensions (exponential function
                    can be expanded in a Taylor series of infinite terms)
                \item When $x = x^\prime$ (identical data points), the kernel is
                    $1$ (maximum similarity):
                    \begin{equation*}
                        K(x,x) = \exp(0) = 1
                    \end{equation*}
                \item As $\Vert x - x^\prime \Vert$ increases, $K(x, x^\prime)
                    \rightarrow 0$
                \item The smaller $\gamma$, the slower $K$ decays (wider
                    influence)
                \item The larger $\gamma$, the faster $K$ decays (narrower
                    influence)
            \end{itemize}
        \end{greenblock}

    \end{columns}
\end{frame}

\begin{frame}{Scaling Input Data}
    \textbf{Scikit-learn \texttt{StandardScaler}}
    \begin{itemize}
        \setlength\itemsep{2em}
        \item Scale data to a mean value of $\mu = 0$ and standard deviation of
            $\sigma = 1$
            \begin{equation*}
                x_\text{scaled} = \frac{x - \mu}{\sigma}
            \end{equation*}
        \item Create a StandardScaler instance: \\
            \texttt{scaler: StandardScaler = StandardScaler()}
        \item Compute the mean and standard deviation for each column for the
            scaling in the next step: \\
            \texttt{scaler.fit(training\_data)}
        \item Actual standardization: \\
            \texttt{training\_data\_scaled = scaler.transform(training\_data)}
    \end{itemize}
\end{frame}

\begin{frame}{Logistic Regression}
    \textbf{Scikit-learn \texttt{LogisticRegression} class}
    \begin{itemize}
        \setlength\itemsep{2em}
        \item Create a model \\
            \texttt{model: LogisticRegression = LogisticRegression()}
        \item Fit the model to the training set \\
            \texttt{model.fit(training\_data, training\_labels)}
        \item Make predictions \\
            \texttt{model.predict(test\_data)}
        \item Test the predictions, for example with the accuracy score \\
            \texttt{metrics.accuracy\_score(test\_labels, predicted\_labels)}
    \end{itemize}
\end{frame}
